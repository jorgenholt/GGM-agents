% Shared style for the thesis figures: colours, TikZ libraries and small drawn icons.
% Loaded once in the preamble of main.tex. Each figure lives in its own file in this
% folder and is pulled into a section with \input{figures/<name>}.
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc,shapes.geometric,shapes.misc,backgrounds,fit,shadows}

\definecolor{ggmSource}{HTML}{C9A227}   % external sources
